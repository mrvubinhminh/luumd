// MÃ GOOGLE APPS SCRIPT DÀNH CHO SMARTLESSON MD PRO
// Cấu trúc chuẩn: Lớp - Tên bài - Chú thích - Chọn nhanh
// =========================================================

const SHEET_NAME = "Lessons";

function setupSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow(["id", "grade", "lessonCode", "title", "note", "quickType", "keywords", "content", "updatedAt"]);
    sheet.getRange("A1:I1").setFontWeight("bold").setBackground("#e0e7ff");
  }
  return sheet;
}

// 1. LẤY DỮ LIỆU TỪ GOOGLE SHEET VỀ ỨNG DỤNG (GET)
function doGet(e) {
  const sheet = setupSheet();
  const rows = sheet.getDataRange().getValues();
  const headers = rows.shift();
  
  const lessons = rows.map(row => {
    // Tương thích ngược với bảng 7 cột cũ và bảng 9 cột mới
    if (row.length >= 9) {
      return {
        id: String(row[0]),
        grade: String(row[1]),
        lessonCode: String(row[2]),
        title: String(row[3]),
        note: String(row[4]),
        quickType: String(row[5]),
        keywords: String(row[6]),
        content: String(row[7]),
        updatedAt: String(row[8])
      };
    } else {
      return {
        id: String(row[0]),
        grade: String(row[1]),
        lessonCode: String(row[2]),
        title: String(row[3]),
        note: "",
        quickType: "",
        keywords: String(row[4]),
        content: String(row[5]),
        updatedAt: String(row[6])
      };
    }
  }).filter(item => item.id && item.title);

  return ContentService.createTextOutput(JSON.stringify({
    status: "success",
    data: lessons
  })).setMimeType(ContentService.MimeType.JSON);
}

// 2. LƯU / CẬP NHẬT / XÓA BÀI HỌC LÊN GOOGLE SHEET (POST)
function doPost(e) {
  try {
    const sheet = setupSheet();
    const payload = JSON.parse(e.postData.contents);
    const action = payload.action;

    if (action === "UPSERT") {
      const item = payload.lesson;
      const data = sheet.getDataRange().getValues();
      let foundRow = -1;
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]) === String(item.id)) {
          foundRow = i + 1;
          break;
        }
      }
      const rowValues = [
        item.id, item.grade, item.lessonCode, item.title, 
        item.note || "", item.quickType || "", item.keywords || "", 
        item.content, item.updatedAt
      ];
      if (foundRow > 0) {
        sheet.getRange(foundRow, 1, 1, 9).setValues([rowValues]);
      } else {
        sheet.appendRow(rowValues);
      }
    } 
    else if (action === "DELETE") {
      const targetId = String(payload.id);
      const data = sheet.getDataRange().getValues();
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]) === targetId) {
          sheet.deleteRow(i + 1);
          break;
        }
      }
    }
    else if (action === "SYNC_ALL") {
      sheet.clearContents();
      sheet.appendRow(["id", "grade", "lessonCode", "title", "note", "quickType", "keywords", "content", "updatedAt"]);
      const lessons = payload.lessons || [];
      if (lessons.length > 0) {
        const rows = lessons.map(item => [
          item.id, item.grade, item.lessonCode, item.title, 
          item.note || "", item.quickType || "", item.keywords || "", 
          item.content, item.updatedAt
        ]);
        sheet.getRange(2, 1, rows.length, 9).setValues(rows);
      }
    }

    return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
                         .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
                         .setMimeType(ContentService.MimeType.JSON);
  }
}
